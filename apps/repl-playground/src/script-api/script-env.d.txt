// Minimal globals available inside playground cells. Loaded only into the
// editor's language service; cells run in a module Web Worker.
interface Console {
  error: (...data: unknown[]) => void
  info: (...data: unknown[]) => void
  log: (...data: unknown[]) => void
  warn: (...data: unknown[]) => void
}
declare const console: Console
declare function setTimeout(handler: () => void, timeout?: number): number
declare function clearTimeout(id: number): void
declare const performance: { now: () => number }

// Script API types by name, so cells can annotate helpers, e.g.
// `function grow(area: Area): Area`. The API file is a module; these aliases
// make its types global for cells only (they are erased before execution).
type Area = import('./auv-api').Area
type AreaEdges = import('./auv-api').AreaEdges
type AreaLength = import('./auv-api').AreaLength
type FocusOptions = import('./auv-api').FocusOptions
type Point = import('./auv-api').Point
type Positional = import('./auv-api').Positional
type Rect = import('./auv-api').Rect

// `auv` is the `@auv-js/sdk` Runner client for the selected Device and the
// current Run, as `createAuv(...).runner(route)` returns in a Node script, and
// `sdk` is the module.
// TODO(playground-sdk-types): typed as `any` until the editor's language service
// loads the SDK's declarations; see
// docs/ai/references/inspect/2026-10-08-playground-sdk-transport-design.md.
// eslint-disable-next-line ts/no-explicit-any
declare const auv: any
// eslint-disable-next-line ts/no-explicit-any
declare const sdk: any
